cmake_minimum_required(VERSION 3.22)
project(AvCamNativeMfCapture LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
add_library(AvCamMfCapture SHARED AvCamMfCapture.cpp)
target_compile_features(AvCamMfCapture PRIVATE cxx_std_17)
target_compile_definitions(AvCamMfCapture PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(AvCamMfCapture PRIVATE mfplat mfreadwrite mf mfuuid ole32)
set_target_properties(AvCamMfCapture PROPERTIES OUTPUT_NAME "AvCamMfCapture")
add_executable(AvCamMfCaptureSmoke AvCamMfCaptureSmoke.cpp)
target_link_libraries(AvCamMfCaptureSmoke PRIVATE AvCamMfCapture)
enable_testing()
add_test(NAME NativeMfCaptureAbi COMMAND AvCamMfCaptureSmoke)
